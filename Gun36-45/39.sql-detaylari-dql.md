## SQL Detayları: DQL

> **Translated to Turkish by** himmetcanumutlu

Şimdi daha önce oluşturduğumuz okul ders seçim sistemi veritabanıyla DML'deki sorgulama işlemini anlatalım. Hem geliştiriciler hem veri analistleri için sorgulama çok önemlidir; ilişkisel veritabanından ihtiyacımız olan veriyi alıp alamamamız buna bağlıdır. Önce iki ders önceki veritabanı ve tablo oluşturan DDL ile bir önceki dersteki veri ekleyen DML'i yeniden çalıştırıp tabloların ve verilerin sorunsuz olduğundan emin olmanızı, sonra aşağıdaki işlemleri yapmanızı öneririz.

```SQL
USE school;

-- Tüm öğrencilerin tüm bilgilerini sorgula
SELECT stu_id,
       stu_name,
       stu_sex,
       stu_birth,
       stu_addr,
       col_id
  FROM tb_student;

-- Öğrencilerin no, ad ve memleketini sorgula (izdüşüm ve takma ad)
SELECT stu_id AS ogrenci_no,
       stu_name AS ad,
       stu_addr AS memleket
  FROM tb_student;

-- Tüm derslerin adını ve kredisini sorgula (izdüşüm ve takma ad)
SELECT cou_name AS ders_adi,
       cou_credit AS kredi
  FROM tb_course;

-- Tüm kız öğrencilerin adını ve doğum tarihini sorgula (veri filtreleme)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0;

-- Memleketi 'Ankara' olan kız öğrencilerin adını ve doğum tarihini sorgula (veri filtreleme)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0
       AND stu_addr = 'Ankara';

-- Memleketi 'Ankara' olan veya kız olan öğrencileri sorgula (veri filtreleme)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0
       OR stu_addr = 'Ankara';

-- Tüm 80'li öğrencilerin ad, cinsiyet ve doğum tarihini sorgula (veri filtreleme)
SELECT stu_name,
       stu_sex,
       stu_birth
  FROM tb_student
 WHERE '1980-1-1' <= stu_birth 
       AND stu_birth <= '1989-12-31';
       
SELECT stu_name,
       stu_sex,
       stu_birth
  FROM tb_student
 WHERE stu_birth BETWEEN '1980-1-1' AND '1989-12-31';

-- Kredisi 2'den büyük derslerin adını ve kredisini sorgula (veri filtreleme)
SELECT cou_name,
	   cou_credit
  FROM tb_course
 WHERE cou_credit > 2;

-- Kredisi tek sayı olan derslerin adını ve kredisini sorgula (veri filtreleme)
SELECT cou_name,
	   cou_credit
  FROM tb_course
 WHERE cou_credit MOD 2 <> 0;

-- 1111 dersini seçip sınav notu 90'dan yüksek olan öğrencilerin numarasını sorgula (veri filtreleme)
SELECT stu_id
  FROM tb_record
 WHERE cou_id = 1111
       AND score > 90;

-- 'Ali Yılmaz' adlı öğrencinin adını ve cinsiyetini sorgula (veri filtreleme)
SELECT stu_name AS ad, 
       CASE stu_sex WHEN 1 THEN 'Erkek' ELSE 'Kadın' END AS cinsiyet
  FROM tb_student
 WHERE stu_name = 'Ali Yılmaz';
 
SELECT stu_name AS ad, 
       IF(stu_sex, 'Erkek', 'Kadın') AS cinsiyet
  FROM tb_student
 WHERE stu_name = 'Ali Yılmaz';
    
-- Soyadı 'Yılmaz' olan öğrencilerin adını ve cinsiyetini sorgula (bulanık eşleştirme)
-- Joker karakter % sıfır veya istenildiği kadar karakteri eşleştirir
SELECT stu_name AS ad, 
       CASE stu_sex WHEN 1 THEN 'Erkek' ELSE 'Kadın' END AS cinsiyet
  FROM tb_student
 WHERE stu_name LIKE 'Yılmaz%';

-- Soyadı 'Yılmaz' olup iki kelimelik adı olan öğrencileri sorgula (bulanık eşleştirme)
-- Joker karakter _ bir karakteri eşleştirir
SELECT stu_name AS ad, 
       CASE stu_sex WHEN 1 THEN 'Erkek' ELSE 'Kadın' END AS cinsiyet
  FROM tb_student
 WHERE stu_name LIKE 'Yılmaz_';

-- Soyadı 'Yılmaz' olup üç kelimelik adı olan öğrencileri sorgula (bulanık eşleştirme)
SELECT stu_name AS ad, 
       CASE stu_sex WHEN 1 THEN 'Erkek' ELSE 'Kadın' END AS cinsiyet
  FROM tb_student
 WHERE stu_name LIKE 'Yılmaz__';
 
-- Öğrenci numarasının son basamağı 3 olan öğrencilerin numarasını ve adını sorgula (bulanık eşleştirme)
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_id LIKE '%3';

-- Adında 'an' veya 'er' geçen öğrencilerin numarasını ve adını sorgula (bulanık eşleştirme ve birleşim)
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%an%'
       OR stu_name LIKE '%er%';
       
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%an%'
 UNION
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%er%';

-- Soyadı 'Yılmaz' veya 'Kaya' olan öğrencilerin numarasını ve adını sorgula (düzenli ifadeyle bulanık eşleştirme)
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name REGEXP '^(Yılmaz|Kaya)';

-- Memleketi girilmemiş öğrencilerin adını sorgula (boş değer işleme)
SELECT stu_name
  FROM tb_student
 WHERE TRIM(stu_addr) = ''
       OR stu_addr is null;
 
-- Memleketi girilmiş öğrencilerin adını sorgula (boş değer işleme)
SELECT stu_name
  FROM tb_student
 WHERE TRIM(stu_addr) <> ''
       AND stu_addr is not null;

-- Öğrencilerin ders seçtiği tüm tarihleri sorgula (yinelenenleri kaldır)
SELECT DISTINCT sel_date
  FROM tb_record;

-- Öğrencilerin memleketlerini sorgula (yinelenenleri kaldır)
SELECT DISTINCT stu_addr
  FROM tb_student
 WHERE TRIM(stu_addr) <> ''
       AND stu_addr is not null;

-- Erkek öğrencilerin adını ve doğum gününü yaşa göre büyükten küçüğe sırala (sıralama)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 1
 ORDER BY stu_birth ASC;
 
-- Ek: yukarıdaki doğum gününü yaşa çevir (tarih fonksiyonu, sayısal fonksiyon)
SELECT stu_name AS ad,
       FLOOR(DATEDIFF(CURDATE(), stu_birth) / 365) AS yas
  FROM tb_student
 WHERE stu_sex = 1
 ORDER BY yas DESC;

-- En büyük yaştaki öğrencinin doğum tarihini sorgula (toplama fonksiyonu)
SELECT MIN(stu_birth)
  FROM tb_student;

-- En küçük yaştaki öğrencinin doğum tarihini sorgula (toplama fonksiyonu)
SELECT MAX(stu_birth)
  FROM tb_student;

-- 1111 numaralı dersin sınav notunun en yükseğini sorgula (toplama fonksiyonu)
SELECT MAX(score)
  FROM tb_record
 WHERE cou_id = 1111;

-- 1001 numaralı öğrencinin sınav notlarının en düşüğünü, en yükseğini, ortalamasını, standart sapmasını, varyansını sorgula (toplama fonksiyonu)
SELECT MIN(score) AS en_dusuk,
       MAX(score) AS en_yuksek,
	   ROUND(AVG(score), 1) AS ortalama,
       STDDEV(score) AS std_sapma,
       VARIANCE(score) AS varyans
  FROM tb_record
 WHERE stu_id = 1001;

-- 1001 numaralı öğrencinin sınav notlarının ortalamasını sorgula; null varsa null 0 sayılır (toplama fonksiyonu)
SELECT ROUND(SUM(score) / COUNT(*), 1) AS ortalama
  FROM tb_record
 WHERE stu_id = 1001;

-- Erkek ve kız öğrenci sayısını sorgula (gruplama ve toplama fonksiyonu)
SELECT CASE stu_sex WHEN 1 THEN 'Erkek' ELSE 'Kadın' END AS cinsiyet,
       COUNT(*) AS kisi
  FROM tb_student
 GROUP BY stu_sex;

-- Her fakültenin öğrenci sayısını sorgula (gruplama ve toplama fonksiyonu)
SELECT col_id AS fakulte_no,
       COUNT(*) AS kisi
  FROM tb_student
 GROUP BY col_id
  WITH ROLLUP;

-- Her fakültedeki erkek ve kız öğrenci sayısını sorgula (gruplama ve toplama fonksiyonu)
SELECT col_id AS fakulte_no,
       CASE stu_sex WHEN 1 THEN 'Erkek' ELSE 'Kadın' END AS cinsiyet,
       COUNT(*) AS kisi
  FROM tb_student
 GROUP BY col_id, stu_sex;

-- Her öğrencinin numarasını ve ortalama notunu sorgula (gruplama ve toplama fonksiyonu)
SELECT stu_id AS ogrenci_no,
	   ROUND(AVG(score), 1) AS ortalama
  FROM tb_record
 GROUP BY stu_id;

-- Ortalama notu 90 ve üzeri olan öğrencilerin numarasını ve ortalamasını sorgula (gruplama sonrası veri filtreleme)
SELECT stu_id AS ogrenci_no,
	   ROUND(AVG(score), 1) AS ortalama
  FROM tb_record
 GROUP BY stu_id
HAVING ortalama >= 90;

-- 1111, 2222, 3333 derslerinin ortalaması 90 ve üzeri olan öğrencilerin numarasını ve ortalamasını sorgula (gruplama öncesi ve sonrası filtreleme)
SELECT stu_id AS ogrenci_no,
	   ROUND(AVG(score), 1) AS ortalama
  FROM tb_record
 WHERE cou_id in (1111, 2222, 3333)
 GROUP BY stu_id
HAVING ortalama >= 90
 ORDER BY ortalama ASC;

-- En büyük yaştaki öğrencinin adını sorgula (alt sorgu)
SELECT stu_name
  FROM tb_student
 WHERE stu_birth = (SELECT MIN(stu_birth)
                      FROM tb_student);

-- İkiden fazla ders seçen öğrencilerin adını sorgula (alt sorgu ve küme işlemi)
SELECT stu_name
  FROM tb_student
 WHERE stu_id in (SELECT stu_id
				    FROM tb_record
				   GROUP BY stu_id
				  HAVING COUNT(*) > 2);

-- Öğrencinin adını, doğum gününü ve bulunduğu fakültenin adını sorgula (tablo birleştirme)
SELECT stu_name,
       stu_birth,
       col_name
  FROM tb_student AS t1, tb_college AS t2
 WHERE t1.col_id = t2.col_id;
 
SELECT stu_name,
       stu_birth,
	   col_name
  FROM tb_student INNER JOIN tb_college
       ON tb_student.col_id = tb_college.col_id;

SELECT stu_name,
       stu_birth,
	   col_name
  FROM tb_student NATURAL JOIN tb_college;
  
SELECT stu_name,
       stu_birth,
	   col_name
  FROM tb_student CROSS JOIN tb_college;

-- Öğrenci adını, ders adını ve notu sorgula (tablo birleştirme)
SELECT stu_name,
       cou_name,
	   score
  FROM tb_student, tb_course, tb_record
 WHERE tb_student.stu_id = tb_record.stu_id
       AND tb_course.cou_id = tb_record.cou_id
       AND score is not null;

SELECT stu_name,
       cou_name,
	   score
  FROM tb_student 
       INNER JOIN tb_record
	       ON tb_student.stu_id = tb_record.stu_id
	   INNER JOIN tb_course
	       ON tb_course.cou_id = tb_record.cou_id
 WHERE score is not null;
 
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null;

-- Ek: yukarıdaki sorgu sonucundan ilk 5 kaydı al (sayfalama sorgusu)
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 5;

-- Ek: yukarıdaki sorgu sonucundan 6-10. kayıtları al (sayfalama sorgusu)
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 5
OFFSET 5;

-- Ek: yukarıdaki sorgu sonucundan 11-15. kayıtları al (sayfalama sorgusu)
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 10, 5;

-- Ders seçen öğrencilerin adını ve ortalama notunu sorgula (alt sorgu ve tablo birleştirme)
-- Error Code: 1248. Every derived table must have its own alias
SELECT stu_name,
	   avg_score
  FROM tb_student
       NATURAL JOIN (SELECT stu_id,
                            ROUND(AVG(score), 1) AS avg_score
                       FROM tb_record
                      GROUP BY stu_id) as tmp;

-- Öğrencilerin adını ve seçtiği ders sayısını sorgula (alt sorgu ve tablo birleştirme)
SELECT stu_name,
	   total
  FROM tb_student
       NATURAL JOIN (SELECT stu_id,
                            COUNT(*) AS total
                       FROM tb_record
                      GROUP BY stu_id) as tmp;

-- Her öğrencinin adını ve ders seçim sayısını sorgula (alt sorgu ve sol dış birleştirme)
SELECT stu_name AS ad,
	   COALESCE(total, 0) AS ders_sayisi
  FROM tb_student AS t1
	   LEFT JOIN (SELECT stu_id,
			             COUNT(*) AS total
		            FROM tb_record
				   GROUP BY stu_id) AS t2
	       ON t1.stu_id = t2.stu_id;
```

Birkaç noktayı açıklamak gerekir:

1. MySQL'in mevcut sürümleri tam dış birleştirmeyi (full outer join) desteklemez; yukarıda `union` işlemiyle sol dış birleştirme ile sağ dış birleştirmenin sonuçlarının birleşimini alarak tam dış birleştirme etkisini elde ettik. 2. MySQL'de birçok operatör tipi desteklenir: aritmetik operatörler (`+`, `-`, `*`, `/`, `%`), karşılaştırma operatörleri (`=`, `<>`, `<=>`, `<`, `<=`, `>`, `>=`, `BETWEEN...AND..`, `IN`, `IS NULL`, `IS NOT NULL`, `LIKE`, `RLIKE`, `REGEXP`), mantıksal operatörler (`NOT`, `AND`, `OR`, `XOR`) ve bitsel operatörler (`&`, `|`, `^`, `~`, `>>`, `<<`); bu operatörleri DML'de veri işlemek için kullanabiliriz.

3. Veri sorgularken, `SELECT` deyimi ve alt cümlelerinde (`WHERE`, `ORDER BY`, `HAVING` gibi) fonksiyonlar kullanılabilir; bu fonksiyonlar string, sayısal, tarih-saat ve akış kontrol fonksiyonlarını içerir; aşağıdaki tablodaki gibidir.

   Yaygın string fonksiyonları.

   | Fonksiyon                        | İşlev                                                  |
   | --------------------------- | ----------------------------------------------------- |
   | `CONCAT`                    | Birden çok string'i tek string'e birleştirir                          |
   | `FORMAT`                    | Sayıyı string'e biçimler ve kaç ondalık tutulacağını belirtir                |
   | `FROM_BASE64` / `TO_BASE64` | BASE64 çözme/kodlama                                       |
   | `BIN` / `OCT` / `HEX`       | Sayıyı ikilik/sekizlik/onaltılık string'e çevirir              |
   | `LOCATE`                    | String'de bir alt string'in konumunu bulur                          |
   | `LEFT` / `RIGHT`            | String'in sol/sağ tarafından belirtilen uzunlukta karakter döndürür                 |
   | `LENGTH` / `CHAR_LENGTH`    | String'in uzunluğunu bayt/karakter cinsinden döndürür                     |
   | `LOWER` / `UPPER`           | String'in küçük/büyük harf biçimini döndürür                             |
   | `LPAD` / `RPAD`             | String yetersizse sol/sağ tarafa belirtilen karakteri doldurur |
   | `LTRIM` / `RTRIM`           | String'in önündeki/arkasındaki boşlukları atar                             |
   | `ORD` / `CHAR`              | Karakterin kodunu / kodun karakterini döndürür                 |
   | `STRCMP`                    | String'leri karşılaştırır; -1, 0, 1 sırasıyla küçük, eşit, büyük demektir      |
   | `SUBSTRING`                 | String'in belirtilen aralıktaki alt string'ini döndürür                              |

   Yaygın sayısal fonksiyonlar.

   | Fonksiyon                                                     | İşlev                               |
   | -------------------------------------------------------- | ---------------------------------- |
   | `ABS`                                                    | Bir sayının mutlak değerini döndürür                 |
   | `CEILING` / `FLOOR`                                      | Bir sayının yukarı/aşağı yuvarlanmış sonucunu döndürür      |
   | `CONV`                                                   | Bir sayıyı bir tabandan başka tabana çevirir |
   | `CRC32`                                                  | Döngüsel artıklık kontrol kodunu hesaplar                 |
   | `EXP` / `LOG` / `LOG2` / `LOG10`                         | Üstel/logaritma hesaplar                      |
   | `POW`                                                    | Üs alır                               |
   | `RAND`                                                   | [0,1) aralığında rastgele sayı döndürür              |
   | `ROUND`                                                  | Bir sayının yuvarlanmış sonucunu döndürür         |
   | `SQRT`                                                   | Bir sayının karekökünü döndürür                 |
   | `TRUNCATE`                                               | Bir sayıyı belirtilen hassasiyete keser             |
   | `SIN` / `COS` / `TAN` / `COT` / `ASIN` / `ACOS` / `ATAN` | Trigonometrik fonksiyonlar                           |

   Yaygın tarih-saat fonksiyonları.

   | Fonksiyon                                      | İşlev                                  |
   | ----------------------------------------- | ------------------------------------- |
   | `CURDATE` / `CURTIME` / `NOW`             | Mevcut tarihi/saati/tarih ve saati alır          |
   | `ADDDATE` / `SUBDATE`                     | İki tarih ifadesini toplar/çıkarır ve sonucu döndürür   |
   | `DATE` / `TIME`                           | String'den tarihi/saati alır               |
   | `YEAR` / `MONTH` / `DAY`                  | Tarihten yılı/ayı/günü alır                  |
   | `HOUR` / `MINUTE` / `SECOND`              | Saatten saati/dakikayı/saniyeyi alır                  |
   | `DATEDIFF` / `TIMEDIFF` / `TIMESTAMPDIFF` | İki tarih-saat ifadesi arasındaki gün/saat farkını döndürür |
   | `MAKEDATE` / `MAKETIME`                   | Bir tarih/saat üretir                     |

   Yaygın akış kontrol fonksiyonları.

   | Fonksiyon     | İşlev                                             |
   | -------- | ------------------------------------------------ |
   | `IF`     | Koşul sağlanıp sağlanmadığına göre farklı değer döndürür                     |
   | `IFNULL` | NULL ise belirtilen değeri, değilse kendisini döndürür           |
   | `NULLIF` | İki ifade eşitse NULL, değilse ilk ifadenin değerini döndürür |

   Diğer yaygın fonksiyonlar.

   | Fonksiyon                       | İşlev                          |
   | -------------------------- | ----------------------------- |
   | `MD5` / `SHA1` / `SHA2`    | String'e karşılık gelen özeti döndürür      |
   | `CHARSET` / `COLLATION`    | Karakter kümesini/sıralama kuralını döndürür           |
   | `USER` / `CURRENT_USER`    | Mevcut kullanıcıyı döndürür                  |
   | `DATABASE`                 | Mevcut veritabanı adını döndürür              |
   | `VERSION`                  | Mevcut veritabanı sürümünü döndürür            |
   | `FOUND_ROWS` / `ROW_COUNT` | Sorgulanan satır sayısını / etkilenen satır sayısını döndürür |
   | `LAST_INSERT_ID`           | Son otomatik artan birincil anahtarın değerini döndürür      |
   | `UUID` / `UUID_SHORT`      | Evrensel benzersiz tanımlayıcı döndürür            |
